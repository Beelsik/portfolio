        -- 1. Ізоляція рядків за користувачем та тенантом
ALTER TABLE public.bookings ENABLE ROW LEVEL SECURITY;

CREATE POLICY "tenant_read_own_bookings"
  ON public.bookings FOR SELECT
  USING (customer_id = auth.uid() OR tenant_admin(tenant_id, auth.uid()));

-- 2. Захист від подвійного бронювання на рівні СУБД
CREATE EXTENSION IF NOT EXISTS btree_gist;
ALTER TABLE public.bookings
  ADD CONSTRAINT prevent_double_booking
  EXCLUDE USING gist (staff_id WITH =, slot_range WITH &&)
  WHERE (status <> 'cancelled');

-- 3. Серверний розрахунок вартості (браузер передає лише ID послуги й слот)
CREATE OR REPLACE FUNCTION public.create_booking(p_service uuid, p_slot tstzrange)
RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE v_price integer; v_id uuid;
BEGIN
  SELECT price_cents INTO STRICT v_price FROM services WHERE id = p_service AND active;
  INSERT INTO bookings (customer_id, service_id, slot_range, total_cents)
  VALUES (auth.uid(), p_service, p_slot, v_price) RETURNING id INTO v_id;
  RETURN v_id;
END; $$;
