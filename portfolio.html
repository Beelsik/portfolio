        On 60k+ LOC codebases, prompt instructions drift. My zero-dependency Node.js `.claude/` kit enforces a plan-first workflow: `PreToolUse` blocks any edit outside approved globs (`exit 2`), protects `.env`/keys/hooks, and `Stop` re-feeds failing Vitest/Playwright logs back to the agent.
        Sur un dépôt de 60 000+ lignes (Electron, React, Node, Supabase), les consignes texte finissent par dériver. Mon kit .claude/ sans dépendances impose : validation du plan, verrouillage des fichiers autorisés par hook PreToolUse (exit 2), blocage des secrets/SSH, et interdiction de terminer tant que les tests échouent.

  USING (customer_id = auth.uid() OR tenant_admin(tenant_id, auth.uid()));

CREATE EXTENSION IF NOT EXISTS btree_gist;
ALTER TABLE public.bookings
  ADD CONSTRAINT prevent_double_booking
  EXCLUDE USING gist (staff_id WITH =, slot_range WITH &&)
  WHERE (status <> 'cancelled');

-- 3. Server-side price calculation (client passes only service ID & slot)
CREATE OR REPLACE FUNCTION public.create_booking(p_service uuid, p_slot tstzrange)
RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE
  v_price integer;
  v_id uuid;
BEGIN
  SELECT price_cents INTO STRICT v_price FROM services WHERE id = p_service AND active;
  INSERT INTO bookings (customer_id, service_id, slot_range, total_cents)
  VALUES (auth.uid(), p_service, p_slot, v_price)
  RETURNING id INTO v_id;
  RETURN v_id;
END;
$$;
